package com.example.cupcakesgourmet;

import android.os.Bundle;
import android.view.LayoutInflater;
import android.view.View;
import android.view.ViewGroup;
import android.widget.Button;
import android.widget.Toast;

import androidx.annotation.NonNull;
import androidx.annotation.Nullable;
import androidx.fragment.app.Fragment;

public class SecondFragment extends Fragment {

    @Nullable
    @Override
    public View onCreateView(
            @NonNull LayoutInflater inflater,
            @Nullable ViewGroup container,
            @Nullable Bundle savedInstanceState) {

        return inflater.inflate(
                R.layout.fragment_second,
                container,
                false
        );
    }

    @Override
    public void onViewCreated(
            @NonNull View view,
            @Nullable Bundle savedInstanceState) {

        super.onViewCreated(view, savedInstanceState);

        Button chocolate = view.findViewById(R.id.btnChocolate);
        Button morango = view.findViewById(R.id.btnMorango);
        Button baunilha = view.findViewById(R.id.btnBaunilha);

        chocolate.setOnClickListener(v ->
                Toast.makeText(
                        requireContext(),
                        "Cupcake de chocolate adicionado!",
                        Toast.LENGTH_SHORT
                ).show()
        );

        morango.setOnClickListener(v ->
                Toast.makeText(
                        requireContext(),
                        "Cupcake de morango adicionado!",
                        Toast.LENGTH_SHORT
                ).show()
        );

        baunilha.setOnClickListener(v ->
                Toast.makeText(
                        requireContext(),
                        "Cupcake de baunilha adicionado!",
                        Toast.LENGTH_SHORT
                ).show()
        );
    }
}
